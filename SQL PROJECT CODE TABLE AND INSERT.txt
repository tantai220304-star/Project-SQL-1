--CREATE TABLE STORE
CREATE TABLE PROJECT_STORES ( 
  store_id     INT PRIMARY KEY, 
  store_name   VARCHAR2(60) NOT NULL, 
  phone        VARCHAR2(20), 
  address      VARCHAR2(120) NOT NULL, 
  city         VARCHAR2(40) NOT NULL, 
  province     CHAR(2) DEFAULT 'ON' NOT NULL, 
  postal_code  VARCHAR2(7) NOT NULL, 
  open_date    DATE DEFAULT SYSDATE NOT NULL, 
  CHECK (REGEXP_LIKE(postal_code, '^[A-Z][0-9][A-Z][ ][0-9][A-Z][0-9]$')), 
  CHECK (province IN ('ON','QC','BC','AB','MB','SK','NS','NB','NL','PE','NT','NU','YT')) 
); 

--CREATE TABLE CUSTSOMERS
CREATE TABLE PROJECT_CUSTOMERS ( 
  customer_id        INT PRIMARY KEY, 
  first_name         VARCHAR2(40) NOT NULL, 
  last_name          VARCHAR2(40) NOT NULL, 
  email              VARCHAR2(80) NOT NULL UNIQUE, 
  phone              VARCHAR2(20), 
  join_date          DATE DEFAULT SYSDATE NOT NULL, 
  pc_optimum_number  VARCHAR2(16) UNIQUE, 
  pc_points_balance  INT DEFAULT 0 NOT NULL, 
  status             VARCHAR2(12) DEFAULT 'ACTIVE' NOT NULL, 
  CHECK (pc_points_balance >= 0), 
  CHECK (status IN ('ACTIVE','INACTIVE')) 

); 

--CREATE TABLE PRODUCTS
CREATE TABLE PROJECT_PRODUCTS ( 
  product_id    NUMBER(7) PRIMARY KEY, 
  product_name  VARCHAR2(80) NOT NULL, 
  category      VARCHAR2(40) NOT NULL, 
  brand         VARCHAR2(40), 
  unit_price    NUMBER(8,2) NOT NULL, 
  stock_qty     NUMBER(6) DEFAULT 0 NOT NULL, 
  sku           VARCHAR2(20) UNIQUE, 
  is_active     CHAR(1) DEFAULT 'Y' NOT NULL, 
  CHECK (unit_price > 0), 
  CHECK (stock_qty >= 0), 
  CHECK (is_active IN ('Y','N')) 
); 

--CREATE TABLE EMPLOYEES
CREATE TABLE PROJECT_EMPLOYEES ( 
  employee_id  NUMBER(7) PRIMARY KEY, 
  store_id     NUMBER(5) NOT NULL, 
  first_name   VARCHAR2(40) NOT NULL, 
  last_name    VARCHAR2(40) NOT NULL, 
  email        VARCHAR2(80) NOT NULL UNIQUE, 
  phone        VARCHAR2(20), 
  job_title    VARCHAR2(40) NOT NULL, 
  hire_date    DATE DEFAULT SYSDATE NOT NULL, 
  salary       NUMBER(9,2), 
  status       VARCHAR2(12) DEFAULT 'ACTIVE' NOT NULL, 
  CHECK (salary IS NULL OR salary >= 0), 
  CHECK (status IN ('ACTIVE','INACTIVE')), 
  FOREIGN KEY (store_id) REFERENCES STORES(store_id) 
); 

--CREATE TABLE ORDERS
CREATE TABLE PROJECT_ORDERS ( 
    order_id NUMBER(8) PRIMARY KEY, 
    customer_id NUMBER(7) NOT NULL, 
    store_id NUMBER(5) NOT NULL, 
    order_date DATE DEFAULT SYSDATE NOT NULL, 
    total_amount NUMBER(10,2) DEFAULT 0 NOT NULL, 
    payment_method VARCHAR2(20) DEFAULT 'Debit' NOT NULL, 
    order_status VARCHAR2(20) DEFAULT 'Completed' NOT NULL, 
    CONSTRAINT chk_orders_total CHECK (total_amount >= 0), 
    CONSTRAINT chk_orders_payment  
        CHECK (payment_method IN ('Cash','Debit','Credit','Online')), 
    CONSTRAINT chk_orders_status  
        CHECK (order_status IN ('Completed','Pending','Cancelled')) 
); 


--CREATE TABLE ODERS_ITEMS
CREATE TABLE PROJECT_ORDER_ITEMS ( 
    order_item_id NUMBER(10) PRIMARY KEY, 
    order_id NUMBER(8) NOT NULL, 
    product_id NUMBER(7) NOT NULL, 
    quantity NUMBER(5) DEFAULT 1 NOT NULL, 
    unit_price NUMBER(8,2) NOT NULL, 
    line_total NUMBER(10,2) NOT NULL, 
    CONSTRAINT chk_orderitems_qty CHECK (quantity > 0), 
    CONSTRAINT chk_orderitems_price CHECK (unit_price >= 0), 
    CONSTRAINT chk_orderitems_total CHECK (line_total >= 0) 
);
